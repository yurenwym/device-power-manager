cmake_minimum_required(VERSION 3.21)
project(WirelessBattery VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build with MSVC on Windows (Visual Studio 2022 C++ Build Tools)")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/utf-8 /W4 /EHsc)
add_library(battery_core STATIC src/hid.cpp src/adapters.cpp)
target_link_libraries(battery_core PUBLIC setupapi hid ole32)
add_executable(WirelessBattery WIN32 src/app.cpp src/settings.cpp src/app.rc)
set_target_properties(WirelessBattery PROPERTIES OUTPUT_NAME "设备电源管理")
target_include_directories(WirelessBattery PRIVATE assets)
target_link_libraries(WirelessBattery PRIVATE battery_core shell32 user32 gdi32 comctl32 uuid advapi32)
target_link_options(WirelessBattery PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
add_executable(BatteryProbe src/probe.cpp)
target_link_libraries(BatteryProbe PRIVATE battery_core)
enable_testing()
add_executable(ProtocolTests tests/protocol_tests.cpp)
add_test(NAME ProtocolTests COMMAND ProtocolTests)
add_executable(IoTests tests/io_tests.cpp)
target_link_libraries(IoTests PRIVATE battery_core)
add_test(NAME IoTests COMMAND IoTests)
